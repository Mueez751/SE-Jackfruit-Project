13def hash_password(password: str) -> str: 

14 """Hash password using PBKDF2-HMAC-SHA256 with unique 16-byte salt (BBMS-SR-003).""" 

15 salt = secrets.token_hex(16) 

16 key = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt.encode("utf-8"), 100_000) 

17 return f"{salt}${key.hex()}"

19def verify_password(password: str, hashed_value: str) -> bool: 

20 """Verify password against salted PBKDF2 hash.""" 

21 try: 

22 salt, expected_hex = hashed_value.split("$", 1) 

23 key = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt.encode("utf-8"), 100_000) 

24 return secrets.compare_digest(key.hex(), expected_hex) 

25 except Exception: 

26 return False
