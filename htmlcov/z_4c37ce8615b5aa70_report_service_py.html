23def generate_donations_report(from_date: Optional[str] = None, to_date: Optional[str] = None) -> List[Dict[str, Any]]: 

24 """Report on donations and collections over a given period (BBMS-F-017).""" 

25 with get_db() as db: 

26 sql = """ 

27 SELECT c.collection_id, c.donor_id, d.full_name as donor_name, c.blood_group, 

28 c.volume_ml, c.collection_date, c.collected_by, u.unit_id, u.status as unit_status 

29 FROM blood_collections c 

30 JOIN donors d ON c.donor_id = d.donor_id 

31 LEFT JOIN blood_units u ON c.collection_id = u.collection_id 

32 WHERE 1=1 

33 """ 

34 params = [] 

35 if from_date: 

36 sql += " AND c.collection_date >= ?" 

37 params.append(from_date) 

38 if to_date: 

39 sql += " AND c.collection_date <= ?" 

40 params.append(to_date) 

41 sql += " ORDER BY c.collection_date DESC" 

42 rows = db.execute(sql, params).fetchall() 

43 return [dict(r) for r in rows]

66def generate_discards_and_expiries_report() -> List[Dict[str, Any]]:
